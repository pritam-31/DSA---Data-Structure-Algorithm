void fnxName(int array[], int size, int key) {
    for (int i = 0; i < size; i++) {
        if (array[i] == key) {
            return i;
        }
    }
    return -1;
}

int main() {
    int array[] = {1, 2, 3, 4, 5};
    int size = sizeof(array) / sizeof(int);

    int key = 3;
    int result = fnxName(array, size, key);

    cout << "Element found at index: " << result << endl;

    return 0;
}
